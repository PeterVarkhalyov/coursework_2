10class AirspaceCollector:

12 

13 Attributes: 

14 nominatim: клиент для получения координат и границ стран.

17 Methods: 

18 collect(): возвращает общий результат сбора в формате словаря.

20 

21 ATTRIBUTION = [

23 "Aircraft data © The OpenSky Network", 

24 ] 

25 SUCCESS_CODE = 0

28 def __init__(self, nominatim: NominatimAPI, opensky: OpenSkyAPI) -> None:

32 def collect(self, countries: list[str], airborne_only: bool = True) -> dict[str, Any]: 

33 """Собрать данные и вернуть ответ об успехе или ошибке.""" 

34 

35 timestamp = int(time.time()) 

36 country_names = [name.strip() for name in countries if name.strip()] 

37 try: 

38 if not country_names: 

39 raise ValueError("Нужно указать хотя бы одну страну") 

40 

41 countries_data: list[dict[str, Any]] = [] 

42 for country_name in country_names: 

43 country = self.nominatim.get_data(country_name) 

44 server_time, aircraft = self.opensky.get_data(country.bounding_box, airborne_only) 

45 countries_data.append( 

46 { 

47 "country": asdict(country), 

48 "data": { 

49 "count": len(aircraft), 

50 "time": server_time, 

51 "aircraft": [asdict(item) for item in aircraft], 

52 }, 

53 } 

54 ) 

55 

56 return { 

57 "attribution": self.ATTRIBUTION.copy(), 

58 "result": True, 

59 "msg": {"code": self.SUCCESS_CODE, "error": None}, 

60 "count": len(countries_data), 

61 "time": timestamp, 

62 "countries": countries_data, 

63 } 

64 except (APIError, ValueError) as error: 

65 return { 

66 "attribution": self.ATTRIBUTION.copy(), 

67 "result": False, 

68 "msg": {"code": self.ERROR_CODE, "error": str(error)}, 

69 "count": 0, 

70 "time": timestamp, 

71 "countries": [], 

72 }
